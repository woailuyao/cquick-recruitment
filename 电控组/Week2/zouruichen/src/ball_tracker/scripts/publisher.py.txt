#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from ball_tracker.msg import BallPos

class BallPublisher(Node):
    def __init__(self):
        super().__init__('ball_publisher')
        self.publisher_ = self.create_publisher(BallPos, 'ball_pos', 10)
        self.timer = self.create_timer(0.5, self.timer_callback) 
        self.balls = [
            BallPos(id=1, x=2.0, y=3.0),
            BallPos(id=2, x=5.5, y=8.0),
            BallPos(id=3, x=9.0, y=4.0)
        ]
        self.index = 0

    def timer_callback(self):
        msg = self.balls[self.index]
        self.publisher_.publish(msg)
        self.get_logger().info(f'发布: 球 ID={msg.id}, X={msg.x}, Y={msg.y}')
        self.index = (self.index + 1) % 3

def main(args=None):
    rclpy.init(args=args)
    node = BallPublisher()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
