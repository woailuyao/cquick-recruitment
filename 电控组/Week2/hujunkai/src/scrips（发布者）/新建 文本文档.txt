#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from ball_interfaces.msg import BallPos

class BallPublisher(Node):
    def __init__(self):
        super().__init__('ball_publisher')
        
        # 话题名按照要求叫 'ball_pos'
        self.publisher_ = self.create_publisher(BallPos, 'ball_pos', 10)
        
        # 要求 2Hz，也就是 0.5 秒触发一次 (如果要改5Hz，把0.5改成0.2)
        self.timer = self.create_timer(0.5, self.timer_callback)
        
        # 题目要求的3个固定坐标（写死）
        self.balls = [
            {'id': 1, 'x': 2.0, 'y': 3.0},
            {'id': 2, 'x': 5.5, 'y': 8.0},
            {'id': 3, 'x': 9.0, 'y': 4.0}
        ]

    def timer_callback(self):
        for ball in self.balls:
            msg = BallPos()
            msg.id = ball['id']
            msg.x = ball['x']
            msg.y = ball['y']
            self.publisher_.publish(msg)
            self.get_logger().info(f'发布: 球号={msg.id}, x={msg.x}, y={msg.y}')

def main(args=None):
    rclpy.init(args=args)
    node = BallPublisher()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()

if __name__ == '__main__':
    main()
