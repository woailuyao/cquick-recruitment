# 写错了不报错、不警告，参数一个都不加载，然后你会以为代码问题查到半夜。
servo_driver:
  ros__parameters:
    port: /dev/ttyUSB0          # 串口设备名（做了 udev 加分项后改成 /dev/cq_servo）

    pwm_min: 900                # 安全下限（结构实测得出，不是电气极限 500）
    pwm_max: 2100               # 安全上限（结构实测得出，不是电气极限 2500）
    default_duration_ms: 800    # 客户端没给 duration 时用，800ms 保证平滑不猛跳
    servo_ids: [101, 102, 103, 104, 105, 106, 107, 108, 109, 110]   # 本机有哪些舵机，用来拒绝不存在的 ID
    startup_reset: true         # 启动时全部 PULR + 回中位 1500
    max_duration_ms: 30000      # 单条指令最大用时上限（防手滑输错）
