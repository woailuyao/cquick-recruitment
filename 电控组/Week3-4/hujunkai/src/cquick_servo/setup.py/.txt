from setuptools import find_packages, setup
import os
from glob import glob

package_name = 'cquick_servo'

# launch/ 和 config/ 默认不会被装进 install/，
# 必须在 data_files 里显式加进去（考核 FAQ Q4 的坑）
data_files = [
    ('share/ament_index/resource_index/packages', ['resource/' + package_name]),
    ('share/' + package_name, ['package.xml']),
    (os.path.join('share', package_name, 'launch'), glob('launch/*.py')),
    (os.path.join('share', package_name, 'config'), glob('config/*.yaml')),
]
action_yamls = glob('config/actions/*.yaml')
if action_yamls:
    data_files.append((os.path.join('share', package_name, 'config', 'actions'), action_yamls))

setup(
    name=package_name,
    version='0.0.1',
    packages=find_packages(exclude=['test']),
    data_files=data_files,
    install_requires=['setuptools'],
    zip_safe=True,
    maintainer='your_name',
    maintainer_email='you@example.com',
    description='总线舵机串口驱动：Action 服务端 / 客户端 / 动作组播放器',
    license='Apache-2.0',
    tests_require=['pytest'],
    entry_points={
        'console_scripts': [
            'servo_driver = cquick_servo.servo_driver:main',
            'move_client = cquick_servo.move_client:main',
            'sequence_player = cquick_servo.sequence_player:main',
            'servo_watchdog = cquick_servo.watchdog:main',
        ],
    },
)
